```bash
_write_provision_history() {
  local rc="${1:-1}"
  local duration=$(( SECONDS - _PROVISION_START ))
  local result="fail"; [ "$rc" -eq 0 ] && result="pass"
  local skill_sha; skill_sha=$(git -C "$SCRIPT_DIR/.." rev-parse --short HEAD 2>/dev/null || echo "unknown")
  local timestamp; timestamp=$(date -u +%Y-%m-%dT%H:%M:%SZ)
  # Resolve history file path relative to coolify.yaml location
  local _yaml_ref="${YAML_PATH:-./coolify.yaml}"
  local _yaml_abs; _yaml_abs="$(cd "$(dirname "$_yaml_ref")" 2>/dev/null && pwd || echo "$(pwd)")"
  local hist_dir="$_yaml_abs/.coolify"
  local hist_file="$hist_dir/provision-history.json"

  mkdir -p "$hist_dir" || return 0  # non-fatal: skip silently on write error

  # Read cleanup suggestions from temp file set before validate.sh call
  local suggestions_raw=""
  if [ -n "${HIST_SUGGESTIONS_FILE:-}" ] && [ -f "${HIST_SUGGESTIONS_FILE:-}" ]; then
    suggestions_raw=$(cat "$HIST_SUGGESTIONS_FILE" 2>/dev/null || echo "")
    rm -f "$HIST_SUGGESTIONS_FILE" || true
  fi

  # Serialize per-env app outcomes to JSON via python3 for the heredoc
  local app_outcomes_json
  app_outcomes_json=$(python3 -c "
import json, sys, os
envs = os.environ.get('_HIST_ENV_NAMES', '').split()
outcomes = {}
for e in envs:
    key = e.upper().replace('-', '_')
    o = os.environ.get('_HIST_APP_OUTCOME_' + key, 'SKIP')
    u = os.environ.get('_HIST_APP_UUID_' + key, '')
    outcomes[e] = {'outcome': o, 'uuid': u} if u else {'outcome': o}
print(json.dumps(outcomes))
" 2>/dev/null || echo "{}")

  python3 - \
    "$hist_file" \
    "$timestamp" "${SERVER_ALIAS:-unknown}" "$skill_sha" "$_yaml_ref" \
    "$result" "$rc" "$duration" \
    "${HIST_PROJECT_OUTCOME:-SKIP}" "${HIST_PROJECT_NAME:-unknown}" \
    "${HIST_VOLUMES_OUTCOME:-SKIP}" \
    "${HIST_TOKENS_OUTCOME:-SKIP}" \
    "${HIST_DNS_OUTCOME:-SKIP}" \
    "$suggestions_raw" \
    "$app_outcomes_json" \
    "$HIST_ISSUES_JSON" \
    <<'PY'
import json, sys, os

hist_file = sys.argv[1]
timestamp, server_alias, skill_sha, yaml_path = sys.argv[2:6]
result = sys.argv[6]
exit_code = int(sys.argv[7])
duration = int(sys.argv[8])
proj_outcome, proj_name = sys.argv[9], sys.argv[10]
vol_outcome, tok_outcome, dns_outcome = sys.argv[11], sys.argv[12], sys.argv[13]
suggestions_raw = sys.argv[14]
app_outcomes_json = sys.argv[15]
issues_json = sys.argv[16] if len(sys.argv) > 16 else "[]"

try:
    app_outcomes = json.loads(app_outcomes_json)
except Exception:
    app_outcomes = {}

resources = {
    "project": {"outcome": proj_outcome, "name": proj_name},
    "volumes": {"outcome": vol_outcome},
    "doppler_tokens": {"outcome": tok_outcome},
    "dns_records": {"outcome": dns_outcome},
}
for env_name, info in app_outcomes.items():
    resources[env_name + "_app"] = info

cleanup_suggestions = []
for line in suggestions_raw.splitlines():
    line = line.strip()
    if not line:
        continue
    try:
        cleanup_suggestions.append(json.loads(line))
    except Exception:
        pass

try:
    issues = json.loads(issues_json)
    if not isinstance(issues, list):
        issues = []
except Exception:
    issues = []

entry = {
    "timestamp": timestamp,
    "server_alias": server_alias,
    "skill_sha": skill_sha,
    "yaml_path": yaml_path,
    "result": result,
    "exit_code": exit_code,
    "duration_seconds": duration,
    "resources": resources,
    "issues": issues,
    "cleanup_suggestions": cleanup_suggestions,
}

CAP = 20
if os.path.exists(hist_file):
    try:
        with open(hist_file) as f:
            history = json.load(f)
        if not isinstance(history, list):
            history = []
    except Exception:
        history = []
else:
    history = []

history.append(entry)
if len(history) > CAP:
    history = history[-CAP:]

with open(hist_file, 'w') as f:
    json.dump(history, f, indent=2)
    f.write('\n')
PY
}

_on_exit() {
  local rc=$?
  # Preserve existing abort-notice behavior
  if [ "$rc" -ne 0 ]; then
    echo "" >&2
    echo "ERROR: provisioning aborted (exit $rc). Coolify may be partially provisioned." >&2
    echo "Re-run /setup-coolify — provisioning is idempotent and resumes safely." >&2
  fi
  # History: skip in plan mode (read-only) and seed mode
  if [ "${PLAN_MODE:-false}" = "true" ]; then return 0; fi
  # Export per-env outcomes for python3 subprocess to read
  local _env_names=""
  for _k in "${!HIST_APP_OUTCOME[@]}"; do
    _env_names="$_env_names $_k"
    local _key; _key="${_k^^}"; _key="${_key//-/_}"
    export "_HIST_APP_OUTCOME_${_key}=${HIST_APP_OUTCOME[$_k]}"
    export "_HIST_APP_UUID_${_key}=${HIST_APP_UUID[$_k]:-}"
  done
  export _HIST_ENV_NAMES="$_env_names"
  _write_provision_history "$rc"
}
trap _on_exit EXIT
```
